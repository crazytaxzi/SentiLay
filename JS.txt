let settings = {
  maxMessages: 7,
  messageLifetime: 26,
  hideCommands: true,
  showBadges: true,
  useModeration: true,
  compactBots: true,
  nameColorMode: "twitch",
  botNames: new Set(),
  blockedUsers: new Set()
};

const chat = document.getElementById("chat");

window.addEventListener("onWidgetLoad", (obj) => {
  const fields = obj.detail.fieldData || {};

  settings = {
    maxMessages: toNumber(fields.maxMessages, 7, 1, 20),
    messageLifetime: toNumber(fields.messageLifetime, 26, 0, 180),
    hideCommands: toBool(fields.hideCommands, true),
    showBadges: toBool(fields.showBadges, true),
    useModeration: toBool(fields.useModeration, true),
    compactBots: toBool(fields.compactBots, true),
    nameColorMode: fields.nameColorMode || "twitch",
    botNames: csvSet(fields.botNames || "streamelements,nightbot,streamlabs,blerp"),
    blockedUsers: csvSet(fields.blockedUsers || "")
  };

  const root = document.documentElement;
  root.style.setProperty("--primary", validColor(fields.primaryAccent, "#d86cff"));
  root.style.setProperty("--secondary", validColor(fields.secondaryAccent, "#4ee7ff"));
  root.style.setProperty("--ember", validColor(fields.emberAccent, "#ff4f9f"));
  root.style.setProperty("--text", validColor(fields.textColor, "#f7f5ff"));
  root.style.setProperty("--muted", validColor(fields.mutedTextColor, "#c8c4d3"));
  const panelAlpha = toNumber(fields.panelOpacity, 38, 0, 100) / 100;
  const glowStrength = toNumber(fields.glowStrength, 62, 0, 100) / 100;
  root.style.setProperty("--panel-alpha", panelAlpha.toFixed(3));
  root.style.setProperty("--panel-alpha-mid", (panelAlpha * 0.94).toFixed(3));
  root.style.setProperty("--panel-alpha-faint", (panelAlpha * 0.30).toFixed(3));
  root.style.setProperty("--bot-alpha", (panelAlpha * 0.68).toFixed(3));
  root.style.setProperty("--bot-alpha-mid", (panelAlpha * 0.55).toFixed(3));
  root.style.setProperty("--bot-alpha-faint", (panelAlpha * 0.16).toFixed(3));
  root.style.setProperty("--blur", `${toNumber(fields.blurAmount, 12, 0, 30)}px`);
  root.style.setProperty("--font-size", `${toNumber(fields.fontSize, 26, 14, 54)}px`);
  root.style.setProperty("--message-gap", `${toNumber(fields.messageGap, 10, 0, 30)}px`);
  root.style.setProperty("--glow-big", `${Math.round(18 * glowStrength)}px`);
  root.style.setProperty("--glow-small", `${Math.round(9 * glowStrength)}px`);

  document.getElementById("chat-shell").style.padding = `${toNumber(fields.outerPadding, 18, 0, 80)}px`;
});

window.addEventListener("onEventReceived", async (obj) => {
  const listener = obj.detail.listener;
  const event = obj.detail.event || {};

  if (listener === "message") {
    await receiveMessage(event.data || {});
    return;
  }

  if (listener === "delete-message") {
    removeByData("msgId", event.msgId ?? event.data?.msgId);
    return;
  }

  if (listener === "delete-messages") {
    removeByData("userId", event.userId ?? event.data?.userId);
  }
});

async function receiveMessage(data) {
  const displayName = String(data.displayName || data.nick || data.tags?.["display-name"] || "Unknown");
  const normalizedName = displayName.trim().toLowerCase();
  const originalText = String(data.text || "");

  if (!originalText.trim()) return;
  if (settings.blockedUsers.has(normalizedName)) return;
  if (settings.hideCommands && originalText.trimStart().startsWith("!")) return;

  let text = originalText;
  let useEmotes = true;

  if (settings.useModeration && window.SE_API?.sanitize) {
    try {
      const result = await SE_API.sanitize({ message: originalText });
      if (result?.skip) return;
      const sanitized = result?.result?.message;
      if (typeof sanitized === "string") {
        text = sanitized;
        useEmotes = sanitized === originalText;
      }
    } catch (_) {
      // Fail open: the message is still HTML-escaped before display.
    }
  }

  const role = getRole(data);
  const isBot = settings.botNames.has(normalizedName);
  const userColor = brightenColor(data.displayColor || data.tags?.color || "#d86cff");
  const nameColor = getNameColor(userColor);
  const message = document.createElement("div");

  message.className = [
    "chat-message",
    `role-${role}`,
    data.isAction ? "is-action" : "",
    isBot && settings.compactBots ? "is-bot" : ""
  ].filter(Boolean).join(" ");

  message.dataset.msgId = String(data.msgId || data.tags?.id || cryptoSafeId());
  message.dataset.userId = String(data.userId || data.tags?.["user-id"] || normalizedName);
  message.style.setProperty("--user", userColor);
  message.style.setProperty("--name-color", nameColor);

  const badges = settings.showBadges ? renderBadges(data.badges || []) : "";
  const body = renderMessageText(text, useEmotes ? (data.emotes || []) : []);

  message.innerHTML = `
    <div class="message-row">
      <span class="meta">
        ${badges ? `<span class="badges">${badges}</span>` : ""}
        <span class="username">${escapeHtml(displayName)}</span>
      </span>
      <span class="message-text">${body}</span>
    </div>`;

  chat.appendChild(message);
  trimOldMessages();

  if (settings.messageLifetime > 0) {
    window.setTimeout(() => removeMessage(message), settings.messageLifetime * 1000);
  }
}

function renderBadges(badges) {
  return badges
    .filter((badge) => badge && badge.url)
    .map((badge) => {
      const title = escapeHtml(badge.description || badge.type || "Badge");
      return `<img class="badge" src="${escapeAttribute(badge.url)}" alt="" title="${title}">`;
    })
    .join("");
}

function renderMessageText(text, emotes) {
  if (!Array.isArray(emotes) || emotes.length === 0) {
    return escapeHtml(text);
  }

  const sorted = [...emotes]
    .filter((emote) => Number.isInteger(emote.start) && Number.isInteger(emote.end))
    .sort((a, b) => a.start - b.start);

  let html = "";
  let cursor = 0;

  for (const emote of sorted) {
    if (emote.start < cursor || emote.start < 0 || emote.end < emote.start) continue;

    html += escapeHtml(text.slice(cursor, emote.start));

    const url = emote.urls?.[4] || emote.urls?.[2] || emote.urls?.[1];
    if (url) {
      html += `<img class="emote" src="${escapeAttribute(url)}" alt="${escapeAttribute(emote.name || "emote")}">`;
    } else {
      html += escapeHtml(text.slice(emote.start, emote.end + 1));
    }

    cursor = emote.end + 1;
  }

  html += escapeHtml(text.slice(cursor));
  return html;
}

function getRole(data) {
  const types = new Set((data.badges || []).map((badge) => String(badge.type || "").toLowerCase()));
  const tags = data.tags || {};

  if (types.has("broadcaster") || tags.badges?.includes("broadcaster")) return "broadcaster";
  if (types.has("moderator") || tags.mod === "1") return "moderator";
  if (types.has("subscriber") || tags.subscriber === "1") return "subscriber";
  return "viewer";
}

function getNameColor(userColor) {
  if (settings.nameColorMode === "accent") return "var(--primary)";
  if (settings.nameColorMode === "white") return "var(--text)";
  return userColor;
}

function trimOldMessages() {
  while (chat.children.length > settings.maxMessages) {
    removeMessage(chat.firstElementChild, true);
  }
}

function removeByData(key, value) {
  if (value === undefined || value === null) return;
  const target = String(value);
  [...chat.children].forEach((node) => {
    if (node.dataset[key] === target) removeMessage(node);
  });
}

function removeMessage(node, immediate = false) {
  if (!node || !node.isConnected || node.classList.contains("message-leaving")) return;

  if (immediate) {
    node.remove();
    return;
  }

  node.classList.add("message-leaving");
  window.setTimeout(() => node.remove(), 580);
}

function csvSet(value) {
  return new Set(String(value)
    .split(",")
    .map((item) => item.trim().toLowerCase())
    .filter(Boolean));
}

function toBool(value, fallback) {
  if (typeof value === "boolean") return value;
  if (value === "true") return true;
  if (value === "false") return false;
  return fallback;
}

function toNumber(value, fallback, min, max) {
  const number = Number(value);
  if (!Number.isFinite(number)) return fallback;
  return Math.min(max, Math.max(min, number));
}

function validColor(value, fallback) {
  return /^#[0-9a-f]{6}$/i.test(String(value || "")) ? value : fallback;
}

function brightenColor(color) {
  if (!/^#[0-9a-f]{6}$/i.test(String(color || ""))) return "#d86cff";

  const r = parseInt(color.slice(1, 3), 16);
  const g = parseInt(color.slice(3, 5), 16);
  const b = parseInt(color.slice(5, 7), 16);
  const luminance = 0.2126 * r + 0.7152 * g + 0.0722 * b;

  if (luminance >= 112) return color;

  const mix = 0.42;
  const rr = Math.round(r + (255 - r) * mix);
  const gg = Math.round(g + (255 - g) * mix);
  const bb = Math.round(b + (255 - b) * mix);
  return `#${[rr, gg, bb].map((part) => part.toString(16).padStart(2, "0")).join("")}`;
}

function escapeHtml(value) {
  return String(value)
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;")
    .replaceAll("'", "&#039;");
}

function escapeAttribute(value) {
  return escapeHtml(value).replaceAll("`", "&#096;");
}

function cryptoSafeId() {
  if (window.crypto?.randomUUID) return crypto.randomUUID();
  return `msg-${Date.now()}-${Math.random().toString(16).slice(2)}`;
}
